APLIKASI: ASLI atau SIMULASI ===== const IS_IOS = /iPad|iPhone|iPod/.test(navigator.userAgent) || (navigator.platform === 'MacIntel' && navigator.maxTouchPoints > 1); // mode 'real' = buka app asli | mode 'sim' = layar simulasi di dalam HP const APPS = { spotify: { mode: 'real', scheme: 'spotify:track:02W2P4Zd9a7LddUMlIH0Bp', web: 'https://open.spotify.com/track/02W2P4Zd9a7LddUMlIH0Bp' }, ig: { mode: 'real', scheme: 'instagram://user?username=unknown_sst', web: 'https://www.instagram.com/unknown_sst/' }, twitter: { mode: 'real', scheme: 'twitter://timeline', web: 'https://x.com/' }, youtube: { mode: 'real', scheme: 'youtube://www.youtube.com/watch?v=FFcgGFpO1n0', web: 'https://youtu.be/FFcgGFpO1n0' }, maps: { mode: 'real', scheme: 'comgooglemaps://?q=Depok', web: 'https://www.google.com/maps/search/Depok' }, oliveyoung: { mode: 'real', web: 'https://global.oliveyoung.com/' }, safari: { mode: 'real', web: 'https://www.google.com/' }, chat: { mode: 'sim' } // ganti ke 'real' + scheme:'kakaotalk://' kalau mau buka KakaoTalk asli }; function launchReal(cfg) { if (IS_IOS && cfg.scheme) { let left = false; const onHide = () => { if (document.hidden) left = true; }; document.addEventListener('visibilitychange', onHide); window.location.href = cfg.scheme; // coba buka app setTimeout(() => { document.removeEventListener('visibilitychange', onHide); if (!left) window.location.href = cfg.web; // app tidak ada -> buka web }, 1500); } else { // Android/desktop: link https otomatis membuka app kalau terpasang window.open(cfg.web, '_blank'); } } function openApp(name) { const cfg = APPS[name] || { mode: 'sim' }; if (cfg.mode === 'real') return launchReal(cfg); const screen = document.getElementById(name + 'App'); if (!screen) { alert('Aplikasi ' + name + ' belum tersedia di simulator.'); return; } closeApp(); screen.style.display = 'flex'; if (name === 'camera') startCamera(); } function closeApp() { document.querySelectorAll('.app-content').forEach(el => el.style.display = 'none'); document.getElementById('photoViewer').style.display = 'none'; stopCamera(); } // ===== Kamera asli ===== let camStream = null, camFacing = 'environment'; async function startCamera() { const video = document.getElementById('cameraFeed'); const status = document.getElementById('camStatus'); stopCamera(); try { camStream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: camFacing }, audio: false }); video.srcObject = camStream; video.play(); status.innerText = camFacing === 'user' ? 'Kamera Depan 🟢' : 'Kamera Belakang 🟢'; } catch (e) { status.innerText = 'Kamera tidak tersedia / izin ditolak'; } } function stopCamera() { if (camStream) { camStream.getTracks().forEach(t => t.stop()); camStream = null; } } function toggleCamera() { camFacing = camFacing === 'user' ? 'environment' : 'user'; startCamera(); }
